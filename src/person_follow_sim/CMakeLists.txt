cmake_minimum_required(VERSION 3.8)
project(person_follow_sim)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(builtin_interfaces REQUIRED)
add_executable(target_simulator src/target_simulator.cpp)
target_include_directories(target_simulator PRIVATE include)
ament_target_dependencies(target_simulator rclcpp rcl_interfaces geometry_msgs nav_msgs std_msgs visualization_msgs)
add_executable(follow_monitor src/follow_monitor.cpp)
ament_target_dependencies(follow_monitor rclcpp geometry_msgs nav_msgs std_msgs)
add_executable(vision_preview_fixture src/vision_preview_fixture.cpp)
ament_target_dependencies(vision_preview_fixture rclcpp rcl_interfaces builtin_interfaces geometry_msgs nav_msgs std_msgs tf2_ros)
install(TARGETS target_simulator follow_monitor vision_preview_fixture DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_scenarios test/test_scenarios.cpp)
  target_include_directories(test_scenarios PRIVATE include)
  add_test(NAME scenarios COMMAND test_scenarios)
endif()
ament_package()
